% Title-page fields. The filter defines \presstitle, \presssubtitle,
% \pressauthor, \pressinstitution, \pressseries, \presslabel, \presstagline and
% \pressedition from `press.cover`. Title, subtitle and author fall back to the
% document's own metadata.

\providecommand*{\@subtitle}{}
\newcommand{\pressifset}[2]{\ifdefempty{#1}{}{#2}}
\newcommand*{\presstitlefield}{\ifdefempty{\presstitle}{\@title}{\presstitle}}
\newcommand*{\presssubtitlefield}{\ifdefempty{\presssubtitle}{\@subtitle}{\presssubtitle}}
\newcommand*{\pressauthorfield}{%
  \ifdefempty{\pressauthor}{{\def\and{\unskip,\space}\@author}}{\pressauthor}}
\newcommand{\pressifsubtitle}[1]{%
  \ifboolexpr{test {\ifdefempty{\presssubtitle}} and test {\ifdefempty{\@subtitle}}}{}{#1}}
\makeatother
